2. Implementacija Korisničkog Interfejsa (Frontend - UI)

Korisnički interfejs u /src direktorijumu nudi čist, moderan i responzivan doživljaj prilagođen i krajnjim korisnicima i DevOps/QA inženjerima.

2.1 Funkcionalni Moduli Aplikacije

  • Autentifikacioni modul (Login / Register): Omogućava prijavu postojećih i registraciju novih korisnika uz klijentsku i serversku validaciju. Uključuje Sandbox Quick-Credentials dugmad za brzo testiranje (@admin, @developer, @reviewer).
  • Katalog filmova (Movie Finder & Reviews): Prikazuje listu filmova sa detaljima (godina, žanr, režiser, ocena, trajanje), pruža filtriranje i pretragu u realnom vremenu, kao i modal za dodavanje i brisanje filmova.
  • Korisnički profil i revizorski logovi (Audit Trail): Prikazuje metapodatke korisnika, omogućava izmenu biografije i lokacije, i nudi uvid u sistemske logove svih akcija (prijava, kreiranje filmova, brisanje, IP adrese).
  • Code Quality & DevOps Hub: Interaktivni editor koji simulira statičku analizu i prikazuje SonarQube ocene (A–E, Quality Gate, tehnički dug, bezbednost), PMD analizu kompleksnosti ($O(N^2)$ petlje) i ESLint stilska pravila.

2.2 Pseudokod: Dinamičko Filtriranje Filmova na Klijentu

ALGORITHM FilterMovies(moviesList, titleQuery, selectedGenre)
  INPUT: 
    moviesList: Array of Movie objects
    titleQuery: String (uneseni tekst pretrage)
    selectedGenre: String (odabrani žanr filter)
  OUTPUT: 
    filteredList: Array of Movie objects

  filteredList ← EMPTY_ARRAY
  normalizedQuery ← TO_LOWERCASE(TRIM(titleQuery))

  FOR EACH movie IN moviesList DO
    matchesTitle ← (normalizedQuery EQUALS "") OR 
                   (TO_LOWERCASE(movie.title) CONTAINS normalizedQuery)
    matchesGenre ← (selectedGenre EQUALS "All") OR 
                   (movie.genre CONTAINS selectedGenre)

    IF matchesTitle AND matchesGenre THEN
      APPEND movie TO filteredList
    END IF
  END FOR

  RETURN filteredList
END ALGORITHM

3. Implementacija Serverskog Dela (Backend & Baza)

Serverski sloj implementiran u server.ts i src/db.ts obezbeđuje REST API komunikaciju, bezbednost podataka i interakciju sa bazom.

3.1 Pseudokod: Kriptografska Autentifikacija Korisnika

ALGORITHM AuthenticateUser(usernameInput, passwordInput)
  INPUT: 
    usernameInput: String, passwordInput: String
  OUTPUT: 
    Result: Object { success: Boolean, user: UserProfile, error: String }

  normalizedUsername ← TO_LOWERCASE(TRIM(usernameInput))
  
  // 1. Dobavljanje korisnika preko parametrizovanog SQL upita (Prevencija SQL Injection)
  userRecord ← DATABASE_QUERY_ONE(
    "SELECT * FROM users WHERE username = ?", 
    [normalizedUsername]
  )

  IF userRecord IS NULL THEN
    RETURN { success: FALSE, error: "User does not exist." }
  END IF

  // 2. Kriptografsko heširanje unete lozinke pomoću SHA-256
  computedHash ← CRYPTO_SHA256(passwordInput)

  // 3. Poredjenje heševa
  IF userRecord.password_hash NOT EQUALS computedHash THEN
    RETURN { success: FALSE, error: "Invalid password." }
  END IF

  // 4. Kreiranje korisničkog profila (bez izlaganja lozinke)
  userProfile ← EXTRACT_PUBLIC_PROFILE(userRecord)
  RETURN { success: TRUE, user: userProfile }
END ALGORITHM

3.2 Primer Koda iz Projekta: Parametrizovani CRUD i Bezbednost (src/db.ts)

// Primer funkcije za bezbedno kreiranje filma sa parametrizovanim upitom
export function createMovie(userId: string, movie: {
  title: string; year: number; genre: string; director: string;
  duration: string; rating: number; cover: string; description: string;
}): Movie {
  const movieId = `mov_${Date.now().toString(36)}`;
  const now = new Date().toLocaleString();

  db.prepare(`
    INSERT INTO movies (id, user_id, title, year, genre, director, duration, rating, cover, description, created_at)
    VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
  `).run(
    movieId, userId, movie.title, movie.year, movie.genre,
    movie.director, movie.duration, movie.rating,
    movie.cover || 'https://images.unsplash.com/photo-1536440136628-849c177e76a1?w=400',
    movie.description, now
  );

  return getMovieById(movieId)!;
}

4. Automatizovano Testiranje i Pokrivenost Testovima

Strategija testiranja obuhvata dva nivoa: Jedinične/Integracione testove (Vitest) i End-to-End browser testove (Playwright).

4.1 Pseudokod: Playwright E2E Test Ciklus (Dodavanje i Brisanje Filma sa Dijalogom)

SCENARIO "E2E: Complete Movie CRUD Lifecycle"
  GIVEN: Browser is launched in Chromium headless mode
  AND: User navigates to BASE_URL

  // 1. Prijava administratora
  WHEN: User clicks "#btn-prefill-admin"
  AND: User clicks "#btn-submit-auth"
  THEN: Page URL redirects to "/" AND Movie Finder dashboard is displayed

  // 2. Otvaranje modala i popunjavanje forme
  WHEN: User clicks "#btn-add-movie-trigger"
  AND: User fills "#movie-title" with "Interstellar Remastered"
  AND: User fills "#movie-rating" with "9.8"
  AND: User clicks "#btn-movie-submit"
  THEN: Modal closes AND new movie card ".test-movie-card" is visible in DOM

  // 3. Pretraga i brisanje uz rukovanje browser popup dijalogom
  WHEN: User fills "#movie-search" with "Interstellar"
  AND: Register dialog listener: ON_DIALOG(d) -> d.accept()
  AND: User clicks ".test-btn-delete-movie" for "Interstellar Remastered"
  THEN: Movie card is removed from DOM AND database row is deleted
END SCENARIO

4.2 Primer Koda iz Projekta: Playwright E2E Implementacija (e2e/auth-and-movies.spec.ts)

test('Complete movie CRUD: Add a new movie, verify it appears, and delete it', async ({ page }) => {
  await page.goto('/');
  await page.locator('#btn-prefill-admin').click();
  await page.locator('#btn-submit-auth').click();
  await expect(page.locator('#movie-search')).toBeVisible({ timeout: 5000 });

  // Otvaranje modala i unos filma
  await page.locator('#btn-add-movie-trigger').click();
  await page.locator('#movie-title').fill('Interstellar Remastered');
  await page.locator('#movie-year').fill('2024');
  await page.locator('#movie-genre').fill('Sci-Fi, Adventure');
  await page.locator('#movie-rating').fill('9.8');
  await page.locator('#btn-movie-submit').click();

  // Verifikacija kreiranog filma
  const movieCard = page.locator('.test-movie-card', { hasText: 'Interstellar Remastered' });
  await expect(movieCard).toBeVisible({ timeout: 4000 });

  // Brisanje filma uz prihvatanje browser konfirmacije
  page.once('dialog', async dialog => {
    expect(dialog.type()).toBe('confirm');
    await dialog.accept();
  });
  await movieCard.locator('.test-btn-delete-movie').click();
  await expect(movieCard).toHaveCount(0);
});

5. Docker i Kontejnerizacija Aplikacije

Optimizovani Multi-Stage Dockerfile rešava problem zavisnosti od C++ alata na Windows-u i obezbeđuje kompaktan produkcioni kontejner.

# ── STAGE 1: Build Faza (Kompajliranje aplikacije) ───────────────
FROM node:20-slim AS builder
RUN apt-get update && apt-get install -y python3 make g++ && rm -rf /var/lib/apt/lists/*
WORKDIR /app
COPY package*.json ./
RUN npm ci
COPY . .
RUN npm run build
RUN npm prune --production

# ── STAGE 2: Produkciono Runtime Okruženje ───────────────────────
FROM node:20-slim AS runner
WORKDIR /app
ENV PORT=3000 NODE_ENV=production
COPY --from=builder /app/dist ./dist
COPY --from=builder /app/node_modules ./node_modules
COPY --from=builder /app/package*.json ./
EXPOSE 3000
CMD ["npm", "start"]

6. CI/CD Pipeline i GitHub Actions Implementacija

Fajl .github/workflows/deploy.yml automatizuje testiranje, lintere, bildovanje i deployment na Google Cloud Run.

6.1 Pseudokod: GitHub Actions CI/CD Pipeline Logika

PIPELINE "Continuous Integration & Continuous Deployment"
  TRIGGER: ON PUSH TO BRANCH "main"

  JOB test-and-lint:
    RUNNER: ubuntu-latest
    STEPS:
      1. CHECKOUT_SOURCE_CODE()
      2. SETUP_NODEJS(version: "20", cache: "npm")
      3. EXECUTE("npm ci")
      4. EXECUTE("npm run lint")           // TypeScript Type-check
      5. EXECUTE("npm run test")           // Vitest Unit Tests
      6. INSTALL_PLAYWRIGHT_BROWSERS()
      7. EXECUTE("npm run test:e2e")       // Playwright E2E Tests
    
    ON FAILURE:
      ABORT_PIPELINE(status: "FAILED", notify: "Developer")

  JOB build-and-deploy:
    NEEDS: test-and-lint                  // STROGI QUALITY GATE
    RUNNER: ubuntu-latest
    STEPS:
      1. AUTHENTICATE_GOOGLE_CLOUD(WorkloadIdentityFederation)
      2. DOCKER_BUILD_IMAGE(
           context: ".",
           cache_from: "type=gha",
           cache_to: "type=gha,mode=max"
         )
      3. DOCKER_PUSH(registry: "europe-west2-docker.pkg.dev", tag: COMMIT_SHA)
      4. CLOUD_RUN_DEPLOY(
           service: "movie-hub-app",
           image: IMAGE_URI,
           region: "europe-west2",
           allow_unauthenticated: TRUE
         )
END PIPELINE

6.2 Primer Koda iz Projekta: GitHub Actions Konfiguracija (.github/workflows/deploy.yml)

name: CI/CD Pipeline - Test, Build & Deploy

on:
  push:
    branches: [ main ]

jobs:
  test-and-lint:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: 'npm'
      - run: npm ci
      - run: npm run lint
      - run: npm run test
      - name: Install Playwright Browsers
        run: npx playwright install --with-deps chromium
      - run: npm run test:e2e

  build-and-deploy:
    needs: test-and-lint
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: docker/setup-buildx-action@v3
      - uses: docker/build-push-action@v5
        with:
          context: .
          push: true
          tags: ${{ steps.meta.outputs.tags }}
          cache-from: type=gha
          cache-to: type=gha,mode=max
      - name: Deploy to Google Cloud Run
        uses: google-github-actions/deploy-cloudrun@v2
        with:
          service: movie-review-hub
          region: europe-west2
          image: ${{ steps.meta.outputs.tags }}

8.1 Primena Page Object Modela (POM) u Playwright Testovima

// e2e/pages/LoginPage.ts
import { Page, Locator } from '@playwright/test';

export class LoginPage {
  readonly page: Page;
  readonly usernameInput: Locator;
  readonly passwordInput: Locator;
  readonly submitButton: Locator;
  readonly adminQuickButton: Locator;

  constructor(page: Page) {
    this.page = page;
    this.usernameInput = page.getByPlaceholder('username');
    this.passwordInput = page.getByPlaceholder('••••••••');
    this.submitButton = page.locator('#btn-submit-auth');
    this.adminQuickButton = page.locator('#btn-prefill-admin');
  }

  async goto() {
    await this.page.goto('/');
  }

  async loginAsAdmin() {
    await this.adminQuickButton.click();
    await this.submitButton.click();
  }
}
